\noindent\textbf{Constants and types.}
We use $\mathit{NumDims}=3$, $\mathit{BitLength}=12$, and
$\mathit{MaxKeyPatternDecryptChecks}=64$.
A point stores $\mathit{Coords}[3]$; a range query stores
$\mathit{Bounds}[3][2]$; a derived pattern key stores
$(\mathit{Pattern}, \mathit{Attrs}, \mathit{Key})$; and a cover candidate
stores $(\mathit{Pattern}, \mathit{CoverageBitset})$.

\begin{algorithm}[t]
\caption{Query decomposition into wildcard patterns}
\begin{algorithmic}[1]
\Procedure{GetCanonicalCover}{$\mathit{min}, \mathit{max}, \mathit{nodeMin}, \mathit{nodeMax}, \mathit{prefix}$}
    \If{$\mathit{min} \le \mathit{nodeMin}$ and $\mathit{nodeMax} \le \mathit{max}$}
        \State \Return $[\mathit{prefix}]$
    \EndIf
    \If{$\mathit{nodeMax} < \mathit{min}$ or $\mathit{nodeMin} > \mathit{max}$}
        \State \Return $[\,]$
    \EndIf
    \State $\mathit{mid} \gets \mathit{nodeMin} + (\mathit{nodeMax}-\mathit{nodeMin})/2$
    \State $\mathit{left} \gets \Call{GetCanonicalCover}{\mathit{min}, \mathit{max}, \mathit{nodeMin}, \mathit{mid}, \mathit{prefix} \Vert 0}$
    \State $\mathit{right} \gets \Call{GetCanonicalCover}{\mathit{min}, \mathit{max}, \mathit{mid}+1, \mathit{nodeMax}, \mathit{prefix} \Vert 1}$
    \State \Return $\mathit{left} \mathbin{+\!\!+} \mathit{right}$
\EndProcedure

\Procedure{CartesianProduct}{$\mathit{dimCovers}$}
    \State $\mathit{result} \gets \mathit{dimCovers}[0]$
    \ForAll{$\mathit{nextCoverList} \in \mathit{dimCovers}[1:]$}
        \State $\mathit{temp} \gets [\,]$
        \ForAll{$\mathit{left} \in \mathit{result}$}
            \ForAll{$\mathit{right} \in \mathit{nextCoverList}$}
                \State append $(\mathit{left} \mathbin{\| \|} \mathit{right})$ to $\mathit{temp}$
            \EndFor
        \EndFor
        \State $\mathit{result} \gets \mathit{temp}$
    \EndFor
    \State \Return $\mathit{result}$
\EndProcedure

\Procedure{MapToIds}{$Q$}
    \State $\mathit{dimCovers} \gets [\,]$
    \ForAll{$d \in \{0,1,2\}$}
        \State $\mathit{minVal} \gets Q.\mathit{Bounds}[d][0]$
        \State $\mathit{maxVal} \gets Q.\mathit{Bounds}[d][1]$
        \State $\mathit{maxDomain} \gets 2^{\mathit{BitLength}} - 1$
        \State append $\Call{GetCanonicalCover}{\mathit{minVal}, \mathit{maxVal}, 0, \mathit{maxDomain}, \epsilon}$ to $\mathit{dimCovers}$
    \EndFor
    \State \Return \Call{CartesianProduct}{$\mathit{dimCovers}$}
\EndProcedure

\Procedure{FormatToWildcardPattern}{$\mathit{prefix}$}
    \State $\mathit{dims} \gets \Call{Split}{\mathit{prefix}, \mathbin{\| \|}}$
    \State $\mathit{pattern} \gets \epsilon$
    \ForAll{$d \in \{0,1,2\}$}
        \State $\mathit{pattern} \gets \mathit{pattern} \Vert \mathit{dims}[d]$
        \State $\mathit{pattern} \gets \mathit{pattern} \Vert \bigl(\ast \text{ repeated } (\mathit{BitLength} - |\mathit{dims}[d]|)\bigr)$
    \EndFor
    \State \Return $\mathit{pattern}$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\caption{Point encoding and ordered subtraction}
\begin{algorithmic}[1]
\Procedure{FormatPointToBinary}{$p$}
    \State $\mathit{bits} \gets \epsilon$
    \ForAll{$d \in \{0,1,2\}$}
        \State $\mathit{bits} \gets \mathit{bits} \Vert \Call{EncodeAsFixedBinary}{p.\mathit{Coords}[d], 12}$
    \EndFor
    \State \Return $\mathit{bits}$
\EndProcedure

\Procedure{Matches}{$\mathit{pattern}, \mathit{pointBits}$}
    \For{$i \gets 0$ to $|\mathit{pattern}|-1$}
        \If{$\mathit{pattern}[i] \ne \ast$ and $\mathit{pattern}[i] \ne \mathit{pointBits}[i]$}
            \State \Return \textbf{false}
        \EndIf
    \EndFor
    \State \Return \textbf{true}
\EndProcedure

\Procedure{GenerateBitOrderCustom}{$\mathit{dimOrder}, \mathit{bitLen}$}
    \State $\mathit{order} \gets [\,]$
    \ForAll{$d \in \mathit{dimOrder}$}
        \For{$b \gets d \cdot \mathit{bitLen}$ to $(d+1)\cdot\mathit{bitLen}-1$}
            \State append $b$ to $\mathit{order}$
        \EndFor
    \EndFor
    \State \Return $\mathit{order}$
\EndProcedure

\Procedure{SubtractPointOrdered}{$\mathit{pattern}, \mathit{pointBits}, \mathit{bitOrder}$}
    \If{\textbf{not} \Call{Matches}{$\mathit{pattern}, \mathit{pointBits}$}}
        \State \Return $[\mathit{pattern}]$
    \EndIf
    \State $\mathit{emptyPatterns} \gets [\,]$
    \State $\mathit{current} \gets$ a mutable copy of $\mathit{pattern}$
    \ForAll{$\mathit{bitIndex} \in \mathit{bitOrder}$}
        \If{$\mathit{current}[\mathit{bitIndex}] = \ast$}
            \State $\mathit{targetBit} \gets \mathit{pointBits}[\mathit{bitIndex}]$
            \State $\mathit{emptyBranch} \gets$ a copy of $\mathit{current}$
            \State $\mathit{emptyBranch}[\mathit{bitIndex}] \gets \Call{Opposite}{\mathit{targetBit}}$
            \State append $\mathit{emptyBranch}$ to $\mathit{emptyPatterns}$
            \State $\mathit{current}[\mathit{bitIndex}] \gets \mathit{targetBit}$
        \EndIf
    \EndFor
    \State \Return $\mathit{emptyPatterns}$
\EndProcedure

\Procedure{SubtractPointsOrdered}{$\mathit{initialPatterns}, \mathit{dataPoints}, \mathit{bitOrder}$}
    \State $\mathit{currentPatterns} \gets \mathit{initialPatterns}$
    \ForAll{$p \in \mathit{dataPoints}$}
        \State $\mathit{pointBits} \gets \Call{FormatPointToBinary}{p}$
        \State $\mathit{nextPatterns} \gets [\,]$
        \ForAll{$\mathit{pattern} \in \mathit{currentPatterns}$}
            \State append all elements of $\Call{SubtractPointOrdered}{\mathit{pattern}, \mathit{pointBits}, \mathit{bitOrder}}$ to $\mathit{nextPatterns}$
        \EndFor
        \State $\mathit{currentPatterns} \gets \mathit{nextPatterns}$
    \EndFor
    \State \Return $\mathit{currentPatterns}$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\caption{Pattern post-processing and attribute encoding}
\begin{algorithmic}[1]
\Procedure{AttributeValueForBit}{$\mathit{slot}, \mathit{bit}$}
    \State $\mathit{payload} \gets \Call{Encode}{\texttt{mhibe}, \lfloor \mathit{slot}/\mathit{BitLength} \rfloor, \mathit{slot} \bmod \mathit{BitLength}, \mathit{bit}}$
    \State \Return \Call{HashToZp}{$\mathit{payload}$}
\EndProcedure

\Procedure{PatternToAttributeList}{$\mathit{pattern}$}
    \State $\mathit{attrs} \gets \{\}$
    \For{$\mathit{slot} \gets 0$ to $|\mathit{pattern}|-1$}
        \If{$\mathit{pattern}[\mathit{slot}] = \ast$}
            \State \textbf{continue}
        \EndIf
        \If{$\mathit{pattern}[\mathit{slot}] \notin \{0,1\}$}
            \State \textbf{error} ``invalid pattern character''
        \EndIf
        \State $\mathit{attrs}[\mathit{slot}] \gets \Call{AttributeValueForBit}{\mathit{slot}, \mathit{pattern}[\mathit{slot}]}$
    \EndFor
    \State \Return $\mathit{attrs}$
\EndProcedure

\Procedure{PatternContainsPattern}{$\mathit{parent}, \mathit{child}$}
    \For{$i \gets 0$ to $|\mathit{parent}|-1$}
        \If{$\mathit{parent}[i] = \ast$}
            \State \textbf{continue}
        \EndIf
        \If{$\mathit{parent}[i] \ne \mathit{child}[i]$}
            \State \Return \textbf{false}
        \EndIf
    \EndFor
    \State \Return \textbf{true}
\EndProcedure

\Procedure{DedupePatterns}{$\mathit{patterns}$}
    \State $\mathit{seen} \gets \varnothing$
    \State $\mathit{unique} \gets [\,]$
    \ForAll{$\mathit{pattern} \in \mathit{patterns}$}
        \If{$\mathit{pattern} \notin \mathit{seen}$}
            \State insert $\mathit{pattern}$ into $\mathit{seen}$
            \State append $\mathit{pattern}$ to $\mathit{unique}$
        \EndIf
    \EndFor
    \State \Return $\mathit{unique}$
\EndProcedure

\Procedure{SelectMaximalPatterns}{$\mathit{patterns}$}
    \State $\mathit{unique} \gets \Call{DedupePatterns}{\mathit{patterns}}$
    \State sort $\mathit{unique}$ by descending number of $\ast$ entries
    \State $\mathit{maximal} \gets [\,]$
    \ForAll{$\mathit{candidate} \in \mathit{unique}$}
        \State $\mathit{dominated} \gets \textbf{false}$
        \ForAll{$\mathit{chosen} \in \mathit{maximal}$}
            \If{\Call{PatternContainsPattern}{$\mathit{chosen}, \mathit{candidate}$}}
                \State $\mathit{dominated} \gets \textbf{true}$
            \EndIf
        \EndFor
        \If{\textbf{not} $\mathit{dominated}$}
            \State append $\mathit{candidate}$ to $\mathit{maximal}$
        \EndIf
    \EndFor
    \State \Return $\mathit{maximal}$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\caption{Approximate minimum key cover}
\begin{algorithmic}[1]
\Procedure{BuildEmptyPointIndex}{$Q, \mathit{realPoints}$}
    \State $\mathit{indexByPoint} \gets \{\}$
    \State $\mathit{nextIndex} \gets 0$
    \ForAll{$p \in \Call{PointsInQuery}{Q}$}
        \If{$p \in \mathit{realPoints}$}
            \State \textbf{continue}
        \EndIf
        \State $\mathit{indexByPoint}[p] \gets \mathit{nextIndex}$
        \State $\mathit{nextIndex} \gets \mathit{nextIndex} + 1$
    \EndFor
    \State \Return $(\mathit{indexByPoint}, \mathit{nextIndex})$
\EndProcedure

\Procedure{BuildCoverCandidates}{$\mathit{patterns}, Q, \mathit{realPoints}$}
    \State $(\mathit{indexByPoint}, \mathit{emptyPointCount}) \gets \Call{BuildEmptyPointIndex}{Q, \mathit{realPoints}}$
    \State $\mathit{candidates} \gets [\,]$
    \ForAll{$\mathit{pattern} \in \mathit{patterns}$}
        \State $\mathit{bounds} \gets \Call{PatternToBounds}{\mathit{pattern}}$
        \State \textbf{assert} $\mathit{bounds} \subseteq Q$
        \State $\mathit{coverage} \gets$ zero bitset of size $\mathit{emptyPointCount}$
        \ForAll{$p \in \mathit{bounds}$}
            \State \textbf{assert} $p \notin \mathit{realPoints}$
            \State $\mathit{pointIndex} \gets \mathit{indexByPoint}[p]$
            \State set bit $\mathit{pointIndex}$ in $\mathit{coverage}$
        \EndFor
        \If{$\mathit{coverage}$ is not empty}
            \State append $(\mathit{Pattern}=\mathit{pattern}, \mathit{CoverageBitset}=\mathit{coverage})$ to $\mathit{candidates}$
        \EndIf
    \EndFor
    \State \Return $(\mathit{candidates}, \mathit{emptyPointCount})$
\EndProcedure

\Procedure{GreedySetCover}{$\mathit{candidates}, \mathit{emptyPointCount}$}
    \State $\mathit{uncovered} \gets$ full bitset of size $\mathit{emptyPointCount}$
    \State $\mathit{selected} \gets [\,]$
    \While{$\mathit{uncovered}$ is not empty}
        \State $\mathit{best} \gets$ the candidate whose coverage intersects $\mathit{uncovered}$ the most
        \State \textbf{assert} $\mathit{best}$ exists
        \State append $\mathit{best}$ to $\mathit{selected}$
        \State $\mathit{uncovered} \gets \mathit{uncovered} - \mathit{best}.\mathit{CoverageBitset}$
    \EndWhile
    \State $\mathit{coverCount}[p] \gets$ number of selected sets covering point $p$
    \Repeat
        \State $\mathit{changed} \gets \textbf{false}$
        \ForAll{$c \in \mathit{selected}$}
            \If{every point covered by $c$ also has cover count at least $2$}
                \State remove $c$ from $\mathit{selected}$
                \State decrement $\mathit{coverCount}$ on all points covered by $c$
                \State $\mathit{changed} \gets \textbf{true}$
            \EndIf
        \EndFor
    \Until{\textbf{not} $\mathit{changed}$}
    \State \Return $\mathit{selected}$ \Comment{Greedy approximation, not a global optimum}
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\caption{WKD-IBE key derivation}
\begin{algorithmic}[1]
\Procedure{DerivePatternKeyFromRoot}{$\mathit{params}, \mathit{rootKey}, \mathit{pattern}$}
    \State $\mathit{attrs} \gets \Call{PatternToAttributeList}{\mathit{pattern}}$
    \State $\mathit{key} \gets \Call{QualifyKey}{\mathit{params}, \mathit{rootKey}, \mathit{attrs}}$
    \State \Return $(\mathit{Pattern}=\mathit{pattern}, \mathit{Attrs}=\mathit{attrs}, \mathit{Key}=\mathit{key})$
\EndProcedure

\Procedure{DerivePatternKeyFromParent}{$\mathit{params}, \mathit{parentKey}, \mathit{childPattern}$}
    \State \textbf{assert} \Call{PatternContainsPattern}{$\mathit{parentKey}.\mathit{Pattern}, \mathit{childPattern}$}
    \State $\mathit{childAttrs} \gets \Call{PatternToAttributeList}{\mathit{childPattern}}$
    \State $\mathit{key} \gets \Call{NonDelegableQualifyKey}{\mathit{params}, \mathit{parentKey}.\mathit{Key}, \mathit{childAttrs}}$
    \State \Return $(\mathit{Pattern}=\mathit{childPattern}, \mathit{Attrs}=\mathit{childAttrs}, \mathit{Key}=\mathit{key})$
\EndProcedure

\Procedure{DeriveInitialPatternKeys}{$\mathit{params}, \mathit{rootKey}, \mathit{initialPatterns}$}
    \State $\mathit{results} \gets [\,]$
    \ForAll{$\mathit{pattern} \in \mathit{initialPatterns}$}
        \State append $\Call{DerivePatternKeyFromRoot}{\mathit{params}, \mathit{rootKey}, \mathit{pattern}}$ to $\mathit{results}$
    \EndFor
    \State \Return $\mathit{results}$
\EndProcedure

\Procedure{DeriveEmptyPatternKeys}{$\mathit{params}, \mathit{initialKeys}, \mathit{emptyPatterns}$}
    \State $\mathit{results} \gets [\,]$
    \ForAll{$\mathit{emptyPattern} \in \mathit{emptyPatterns}$}
        \State find $\mathit{parentKey} \in \mathit{initialKeys}$ such that
        \Statex \hspace{\algorithmicindent} \Call{PatternContainsPattern}{$\mathit{parentKey}.\mathit{Pattern}, \mathit{emptyPattern}$}
        \If{no such $\mathit{parentKey}$ exists}
            \State \textbf{error} ``missing canonical-cover parent''
        \EndIf
        \State append $\Call{DerivePatternKeyFromParent}{\mathit{params}, \mathit{parentKey}, \mathit{emptyPattern}}$ to $\mathit{results}$
    \EndFor
    \State \Return $\mathit{results}$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\caption{Completeness and cryptographic verification}
\begin{algorithmic}[1]
\Procedure{CalculateVolume}{$\mathit{pattern}$}
    \State \Return $2^{(\text{number of } \ast \text{ in } \mathit{pattern})}$
\EndProcedure

\Procedure{PatternToBounds}{$\mathit{pattern}$}
    \ForAll{dimensions encoded in $\mathit{pattern}$}
        \State interpret $0$ as a fixed $0$, $1$ as a fixed $1$, and $\ast$ as $[0,1]$
    \EndFor
    \State \Return the resulting bounds
\EndProcedure

\Procedure{VerifyEmptyRegionPatterns}{$Q, \mathit{emptyPatterns}, \mathit{realPoints}$}
    \State $\mathit{coverage} \gets$ map from spatial point to hit count
    \State $\mathit{maxOverlap} \gets 0$
    \ForAll{$\mathit{pattern} \in \mathit{emptyPatterns}$}
        \State $\mathit{bounds} \gets \Call{PatternToBounds}{\mathit{pattern}}$
        \State \textbf{assert} $\mathit{bounds} \subseteq Q$
        \ForAll{$p \in \mathit{bounds}$}
            \State \textbf{assert} $p \notin \mathit{realPoints}$
            \State $\mathit{coverage}[p] \gets \mathit{coverage}[p] + 1$
            \State $\mathit{maxOverlap} \gets \max(\mathit{maxOverlap}, \mathit{coverage}[p])$
        \EndFor
    \EndFor
    \State $\mathit{coveredEmptyPoints} \gets 0$
    \ForAll{$p \in \Call{PointsInQuery}{Q}$}
        \If{$p \in \mathit{realPoints}$}
            \State \textbf{assert} $\mathit{coverage}[p] = 0$
        \Else
            \State \textbf{assert} $\mathit{coverage}[p] \ge 1$
            \State $\mathit{coveredEmptyPoints} \gets \mathit{coveredEmptyPoints} + 1$
        \EndIf
    \EndFor
    \State \textbf{assert} $\mathit{coveredEmptyPoints} \ge |\mathit{coverage}|$
    \State \Return $(\mathit{coveredEmptyPoints}, \mathit{maxOverlap})$
\EndProcedure

\Procedure{VerifyDerivedPatternKeys}{$\mathit{params}, \mathit{derivedKeys}$}
    \State $\mathit{checks} \gets \min(|\mathit{derivedKeys}|, \mathit{MaxKeyPatternDecryptChecks})$
    \For{$\mathit{probe} \gets 0$ to $\mathit{checks}-1$}
        \State $\mathit{idx} \gets \mathit{probe} \cdot |\mathit{derivedKeys}| / \mathit{checks}$
        \State $\mathit{item} \gets \mathit{derivedKeys}[\mathit{idx}]$
        \State $\mathit{message} \gets \Call{RandomEncryptable}{}$
        \State $\mathit{ciphertext} \gets \Call{Encrypt}{\mathit{message}, \mathit{params}, \mathit{item}.\mathit{Attrs}}$
        \State $\mathit{decrypted} \gets \Call{Decrypt}{\mathit{ciphertext}, \mathit{item}.\mathit{Key}}$
        \State \textbf{assert} $\mathit{decrypted} = \mathit{message}$
    \EndFor
    \State \Return $\mathit{checks}$ \Comment{Checks same-pattern encryption/decryption}
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[t]
\caption{Main benchmark pipeline}
\begin{algorithmic}[1]
\Procedure{Main}{}
    \State print ``Starting benchmark''
    \State initialize the curve
    \State $(\mathit{params}, \mathit{masterKey}) \gets \Call{Setup}{36, \textbf{true}}$
    \State load accumulator keys
    \State define query $Q$ over date, discount, and quantity
    \State $\mathit{dbData}, \mathit{dbFr}, I, X \gets [\,], [\,], [\,], [\,]$
    \State $\mathit{queryUniquePoints} \gets \varnothing$
    \ForAll{lines in the TPC-H file}
        \State $p \gets$ parse the 3-dimensional point
        \State append $p$ to $\mathit{dbData}$
        \State $\mathit{fr} \gets \Call{SeedToFr}{\Call{FormatPointToBinary}{p}}$
        \State append $\mathit{fr}$ to $\mathit{dbFr}$
        \If{$p \in Q$}
            \State append $\mathit{fr}$ to $I$
            \State insert $p.\mathit{Coords}$ into $\mathit{queryUniquePoints}$
        \Else
            \State append $\mathit{fr}$ to $X$
        \EndIf
    \EndFor
    \State $\mathit{digestDB} \gets \Call{CommitFakeG1}{\mathit{dbFr}}$
    \State $\mathit{digestX} \gets \Call{CommitFakeG1}{X}$
    \State $\mathit{initialPrefixes} \gets \Call{MapToIds}{Q}$
    \State $\mathit{initialPatterns} \gets$ map each prefix in $\mathit{initialPrefixes}$ through \Call{FormatToWildcardPattern}{}
    \State $\mathit{permutations} \gets$ all 6 dimension orderings
    \State $\mathit{combinedEmptyPatterns} \gets [\,]$
    \ForAll{$\pi \in \mathit{permutations}$}
        \State $\mathit{order} \gets \Call{GenerateBitOrderCustom}{\pi, \mathit{BitLength}}$
        \State $\mathit{patterns} \gets \Call{SubtractPointsOrdered}{\mathit{initialPatterns}, \mathit{dbData}, \mathit{order}}$
        \State append all elements of $\mathit{patterns}$ to $\mathit{combinedEmptyPatterns}$
    \EndFor
    \State $\mathit{maximalEmptyPatterns} \gets \Call{SelectMaximalPatterns}{\mathit{combinedEmptyPatterns}}$
    \State $(\mathit{coverCandidates}, \mathit{emptyPointCount}) \gets \Call{BuildCoverCandidates}{\mathit{maximalEmptyPatterns}, Q, \mathit{queryUniquePoints}}$
    \State $\mathit{selectedCover} \gets \Call{GreedySetCover}{\mathit{coverCandidates}, \mathit{emptyPointCount}}$
    \State $\mathit{selectedCoverPatterns} \gets$ extract the pattern field from each item in $\mathit{selectedCover}$
    \State $\mathit{rootKey} \gets \Call{KeyGen}{\mathit{params}, \mathit{masterKey}, \varnothing}$
    \State $\mathit{initialKeys} \gets \Call{DeriveInitialPatternKeys}{\mathit{params}, \mathit{rootKey}, \mathit{initialPatterns}}$
    \State $\mathit{verificationEmptyKeys} \gets \Call{DeriveEmptyPatternKeys}{\mathit{params}, \mathit{initialKeys}, \mathit{selectedCoverPatterns}}$
    \State $\mathit{totalKeyBytes} \gets$ sum of serialized sizes of all verification keys
    \State $\mathit{totalQueryVolume} \gets \sum_{\mathit{pattern} \in \mathit{initialPatterns}} \Call{CalculateVolume}{\mathit{pattern}}$
    \State $(\mathit{coveredEmptyPoints}, \mathit{maxOverlap}) \gets \Call{VerifyEmptyRegionPatterns}{Q, \mathit{selectedCoverPatterns}, \mathit{queryUniquePoints}}$
    \State $\mathit{sampledKeyChecks} \gets \Call{VerifyDerivedPatternKeys}{\mathit{params}, \mathit{verificationEmptyKeys}}$
    \State $\mathit{realSpatialVolume} \gets |\mathit{queryUniquePoints}|$
    \State \textbf{assert} $\mathit{coveredEmptyPoints} + \mathit{realSpatialVolume} = \mathit{totalQueryVolume}$
    \State $\mathit{IPoly} \gets \Call{PolyTree}{I}$
    \State $\mathit{CI} \gets$ Pedersen commitment of $\mathit{IPoly}$
    \State $\mathit{zkMemProof} \gets \Call{ZKMemProver}{\mathit{CI}, \mathit{digestX}, \mathit{transcript}}$
    \State $\mathit{zkDegProof} \gets \Call{ZKDegCheckProver}{\mathit{CI}, \mathit{IPoly}, \Call{Hash}{\mathit{zkMemProof}}}$
    \State $\mathit{ok1} \gets \Call{ZKMemVerifier}{\mathit{zkMemProof}, \mathit{digestDB}, \mathit{CI}.\mathit{Com}, \mathit{transcript}}$
    \State $\mathit{ok2} \gets \Call{ZKDegCheckVerifier}{\mathit{CI}.\mathit{Com}, \mathit{zkDegProof}, \Call{Hash}{\mathit{zkMemProof}}}$
    \State print the final timing report
\EndProcedure
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Pipeline summary.}
First decompose the 3-dimensional query box into canonical prefix blocks.
Then run six dimension orders to extract empty-region patterns.
Next deduplicate the six outputs and keep only maximal empty blocks under pattern containment.
Then compute a greedy approximate minimum key cover on those maximal blocks.
Materialize the selected cover as WKD-IBE completeness keys, verify their geometric union over the empty region, sample same-pattern encrypt/decrypt checks, and finally prove authenticity of the hit set with the ZK accumulator.
